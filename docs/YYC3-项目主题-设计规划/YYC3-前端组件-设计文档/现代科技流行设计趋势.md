#### 粒子系统

**浮动粒子**:
```javascript
class ParticleSystem {
  constructor(container, count = 30) {
    this.container = container;
    this.particles = [];
    this.initParticles(count);
    this.animate();
  }

  initParticles(count) {
    for (let i = 0; i < count; i++) {
      const particle = {
        x: Math.random() * window.innerWidth,
        y: Math.random() * window.innerHeight,
        size: Math.random() * 4 + 1,
        speedX: (Math.random() - 0.5) * 2,
        speedY: (Math.random() - 0.5) * 2,
        opacity: Math.random() * 0.5 + 0.2
      };
      this.particles.push(particle);
    }
  }

  animate() {
    this.particles.forEach(p => {
      p.x += p.speedX;
      p.y += p.speedY;

      if (p.x < 0 || p.x > window.innerWidth) p.speedX *= -1;
      if (p.y < 0 || p.y > window.innerHeight) p.speedY *= -1;
    });

    requestAnimationFrame(() => this.animate());
  }
}
```

**效果**:
- 30+ 浮动粒子
- 随机运动轨迹
- 动态透明度

### 6. 苹果风格交互

#### 弹簧动画（Spring Animation）

**物理模拟**:
```javascript
class SpringAnimation {
  constructor(element, stiffness = 0.1, damping = 0.8) {
    this.element = element;
    this.stiffness = stiffness;
    this.damping = damping;
    this.velocity = 0;
    this.target = 0;
    this.current = 0;
  }

  animate() {
    const force = (this.target - this.current) * this.stiffness;
    this.velocity += force;
    this.velocity *= this.damping;
    this.current += this.velocity;

    this.element.style.transform = `translateY(${this.current}px)`;

    if (Math.abs(this.velocity) > 0.01) {
      requestAnimationFrame(() => this.animate());
    }
  }

  setTarget(value) {
    this.target = value;
    this.animate();
  }
}
```

**效果**:
- 自然的物理运动
- 弹性回弹效果
- 平滑过渡
- 可调节参数

### 8. 数据可视化与科技感

#### 实时数据流

**动态图表**:
```javascript
class RealTimeChart {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d');
    this.data = [];
    this.maxPoints = 100;
    this.animate();
  }

  addData(value) {
    this.data.push(value);
    if (this.data.length > this.maxPoints) {
      this.data.shift();
    }
  }

  animate() {
    this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

    // 绘制渐变填充
    const gradient = this.ctx.createLinearGradient(0, 0, 0, this.canvas.height);
    gradient.addColorStop(0, 'rgba(102, 126, 234, 0.8)');
    gradient.addColorStop(1, 'rgba(102, 126, 234, 0.1)');

    this.ctx.fillStyle = gradient;
    this.ctx.beginPath();
    this.ctx.moveTo(0, this.canvas.height);

    this.data.forEach((value, index) => {
      const x = (index / this.maxPoints) * this.canvas.width;
      const y = this.canvas.height - (value * this.canvas.height);
      this.ctx.lineTo(x, y);
    });

    this.ctx.lineTo(this.canvas.width, this.canvas.height);
    this.ctx.closePath();
    this.ctx.fill();

    // 绘制发光线条
    this.ctx.strokeStyle = '#667eea';
    this.ctx.lineWidth = 2;
    this.ctx.shadowColor = '#667eea';
    this.ctx.shadowBlur = 10;
    this.ctx.stroke();

    requestAnimationFrame(() => this.animate());
  }
}
```

**图片懒加载**:
```javascript
class LazyLoader {
  constructor() {
    this.observer = new IntersectionObserver(this.onIntersect, {
      rootMargin: '50px'
    });
    this.observeElements();
  }

  observeElements() {
    document.querySelectorAll('[data-lazy]').forEach(el => {
      this.observer.observe(el);
    });
  }

  onIntersect(entries) {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        const element = entry.target;
        const src = element.dataset.src;
        element.src = src;
        element.classList.add('loaded');
        this.observer.unobserve(element);
      }
    });
  }
}
```
